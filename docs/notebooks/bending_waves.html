#timescales: the wave's group speed |c_g| = pi G sigma/nu_a (Eqn. 26), and its crossing time tau_c from r_V
#to the ring's inner edge

c_g = np.pi*sigma/nu_V
crossing_time = (sat.r_V - 1.0)/c_g
times_crossing = o['times']/crossing_time
orbits = o['times']/(2.0*np.pi)
print('crossing_time (orbits) = ', np.round(crossing_time/(2.0*np.pi), 1))
print('run length (orbits) = ', np.round(orbits[-1], 1), '  = ', np.round(times_crossing[-1], 3), ' tau_c')

#test 1: each streamline's inclination at the end versus linear theory, |h|/r of Eqn. (34), and the forced
#inclination |L_V/x|/r of Eqn. (35b) alone
A_V = sat.mass_final*sat.A_V_per_mass
Z_w = A_V*np.sqrt(sat.r_V/(sigma*sat.D_V))
x_now = ((o['a'][-1] - sat.r_V)/sat.r_V).mean(axis=1)
inc_now = o['inc'][-1].mean(axis=1)
xs = np.linspace(x.min(), x.max(), 20000)
inc_theory = np.abs(bending_wave(sat, sigma, xs))/(sat.r_V*(1.0 + xs))
inc_forced = A_V/(sat.D_V*np.abs(xs))/(sat.r_V*(1.0 + xs))
f, ax = plt.subplots(figsize=(10, 6))
p1 = ax.plot(x_now, inc_now/1.0e-6, marker='o', markersize=3, linestyle='none', color=color_seq[0], label='simulation')
p2 = ax.plot(xs, inc_theory/1.0e-6, linestyle='dashed', linewidth=2, color=color_seq[0], label='linear theory')
p3 = ax.plot(xs, inc_forced/1.0e-6, linestyle='dotted', linewidth=2, color=color_seq[1], label='forced response')
ax.axvline(x_front, color='gray', linewidth=1, linestyle='dotted')
ax.set_xlim(x.min(), x.max())
ax.set_ylim(0.0, 1.9*Z_w/sat.r_V/1.0e-6)
ax.set_xlabel(r'fractional distance from resonance $x$')
ax.set_ylabel(r'inclination $i$    ($10^{-6}$)')
ax.legend(loc='upper right')
behind = (x < x_hi) & (x > x_lo)
h_wave = h - 1j*A_V/(sat.D_V*x)
print('behind the front: median |h - L_V/x|/|Z_w| = ', np.round(np.median(np.abs(h_wave[behind]))/Z_w, 4))
h_theory = bending_wave(sat, sigma, x)
outside = x > 2.0*np.sqrt(2.0*np.pi*sigma/(sat.r_V*sat.D_V))
print('nonwave side (x > 2 eps^1/2): largest |h - theory|/|theory| = ',
    np.round((np.abs(h - h_theory)/np.abs(h_theory))[outside].max(), 4))
